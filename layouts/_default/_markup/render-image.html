{{- /* Page-resource images become resized webp with a 720w srcset; remote and non-resource images pass through. */ -}} {{- $dest := .Destination -}} {{- $img := "" -}} {{- if and $dest (not (hasPrefix $dest "http")) (not (hasPrefix $dest "/")) (not (hasPrefix $dest "data:")) -}} {{- with .Page.Resources.Get $dest }}{{ if in (slice "image/png" "image/jpeg") .MediaType.Type }}{{ $img = . }}{{ end }}{{ end -}} {{- end -}} {{- if $img -}} {{- $n := .Page.Store.Get "imgn" | default 0 -}} {{- .Page.Store.Set "imgn" (add $n 1) -}} {{- $big := $img -}}{{- if gt $img.Width 1200 }}{{ $big = $img.Resize "1200x webp q80" }}{{ else }}{{ $big = $img.Resize (printf "%dx webp q80" $img.Width) }}{{ end -}} {{- $small := $img.Resize "720x webp q80" -}} {{- else -}} {{- end -}}
